function laneY(idx){ return LANES[idx]; }
function randLane(){ return Math.floor(Math.random()*LANES.length); }
function makeUnit(team,key,x,lane){
  const d=UNIT_DEFS[key];
  const hero = !!d.hero;
  return {
    id:Math.random().toString(36).slice(2), team, key, lane, x, y:laneY(lane), baseY:laneY(lane), hp:d.hp, maxHp:d.hp,
    dmg:d.dmg, speed:d.speed, range:d.range, cd:d.cd, attackCd:Math.random()*.2, size:d.size, row:d.row, projectile:!!d.projectile,
    state:'walk', anim:Math.random(), facing:team==='player'?1:-1, dead:false, death:0, hit:0, hero, aura: hero?1:0,
    buff:0, push: team==='player' ? 1 : -1, recoil:0, zOffset:0
  };
}

function spawnPlayer(key){
  if(!state.started || state.paused) return;
  const d=UNIT_DEFS[key];
  if(state.mana < d.cost){ showToast('Not enough mana'); return; }
  if(state.cooldowns[key] > 0){ showToast('Unit recharging'); return; }
  state.mana -= d.cost; state.cooldowns[key] = key==='cavalry'?1.8:.9;
  const lane=randLane();
  const u=makeUnit('player',key,130+Math.random()*30,lane);
  state.units.push(u);
  addLog(`${d.name} deployed.`);
}

function spawnHero(){
  if(!state.started || state.paused) return;
  if(state.heroCooldown>0){ showToast('Arthos is regrouping'); return; }
  const count = state.units.filter(u=>u.hero&&!u.dead).length;
  if(count>0){ showToast('Arthos is already in battle'); return; }
  state.heroCooldown=26;
  const u=makeUnit('player','arthos',118,1);
  u.hp = u.maxHp = 720; u.dmg=58; u.size=2.3; u.aura=1;
  state.units.push(u);
  flash(.7); addLog('Arthos rides into the fray!');
}

function meteorRain(){
  if(!state.started || state.paused) return;
  if(state.mana<180){ showToast('Need 180 mana'); return; }

  }

}

function ballistaBarrage(){

function spawnEnemy(){
  if(!state.queue.length) return;
  const key = state.queue.shift();
  const lane = randLane();
  const u = makeUnit('enemy',key,1455+Math.random()*25,lane);
  state.units.push(u);
}

function aliveUnits(team){ return state.units.filter(u=>u.team===team && !u.dead); }

function nearestEnemy(u){
  let best=null, bestD=99999;
  for(const e of state.units){
    if(e.dead || e.team===u.team) continue;
    const dx=Math.abs(e.x-u.x); const dy=Math.abs(e.baseY-u.baseY);
    if(dy>55) continue;
    if(dx<bestD){ bestD=dx; best=e; }
  }
  return [best,bestD];
}

function hitTarget(attacker,target,amount,crit=false){
  target.hp -= amount;
  target.hit=.14; target.recoil = (attacker.team==='player'?1:-1)*(attacker.key==='cavalry'||attacker.key==='raider'?16:9);
  spawnBurst(target.x, target.y-90, crit?22:14, attacker.team==='player'?'#93d5ff':'#ff8e8e');
  if(target.hp<=0 && !target.dead){
    target.dead=true; target.death=.95; target.state='dead'; state.kills += attacker.team==='player' ? 1 : 0; state.gold += attacker.team==='player' ? 8 : 0;
    if(target.hero) addLog('A champion has fallen.');
  }
}

function spawnBurst(x,y,count,color){
  for(let i=0;i<count;i++) state.particles.push({x,y,vx:(Math.random()*2-1)*130,vy:-30-Math.random()*180,life:.35+Math.random()*.35,size:2+Math.random()*4,color});
}
